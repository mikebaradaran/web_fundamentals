    <span id="rangeValue"></span>
    <script>
        function highlightBigNumbersGreaterThan(x) {
            for (let e of document.querySelectorAll('.tableRow div')) {
                let no = Number(e.innerHTML);
                e.classList.remove("bigNumber");
                if (no > x)
                    e.classList.add("bigNumber");
            }
        }

        function rangeHighlight(v) {
            document.getElementById("rangeValue").innerHTML = v;
            highlightBigNumbersGreaterThan(v);
        }

        function multi(min, max) {
            var table = "<div>";
            for (var row = min; row <= max; row++) {
                table += "<div class='tableRow'>";
                for (var column = min; column <= max; column++) {
                    table += `<div>${row * column}</div>`;
                }
                table += "</div>";
            }
            table += "</div>"
            return table;
        }

        let tbl = multi(1, 10);
        document.getElementById("myTable").innerHTML = tbl;
        highlightBigNumbersGreaterThan(60);

    </script>
